{{- /* Settings popover in the navbar: theme toggle + language flags. Visibility is hover/focus-driven on desktop. The [data-theme-toggle] checkbox is the contract that assets/js/shared/theme.js looks for. Each language flag links to the equivalent page in that language. Hugo populates .Translations on every page; if a translation is missing for the current page, we fall back to that language's home page so the link is never dead. */ -}} {{- $currentLang := site.Language.Lang -}}
{{ i18n "theme" }}
{{ i18n "language" }}
{{- /* For each configured language, render its flag. The active language renders as a static span; others link to the current page's translation if it exists, else to that language's homepage. */ -}} {{- $defaultLang := site.Home.Site.Language.Lang -}} {{- range site.Languages }} {{- $lang := .Lang -}} {{- $isCurrent := eq $lang $currentLang -}} {{- $href := cond (eq $lang $defaultLang) "/" (printf "/%s/" $lang) -}} {{- with index (where $.Translations "Language.Lang" $lang) 0 -}} {{- $href = .RelPermalink -}} {{- end -}} {{- $aria := i18n (printf "lang_%s" (cond (eq $lang "en") "english" "danish")) -}} {{- if $isCurrent }} {{- partial "settings/flag.html" $lang -}} {{- else }} {{- partial "settings/flag.html" $lang -}} {{- end }} {{- end }}
