{{- /* Language switcher. Globe-icon button reveals a small popover containing just the language flags — much narrower than the old combined settings panel, so it doesn't clip on mobile. Each language flag links to the equivalent page in that language; if a translation is missing for the current page, we fall back to that language's home page so the link is never dead. */ -}} {{- $currentLang := site.Language.Lang -}}
{{- $defaultLang := site.Home.Site.Language.Lang -}} {{- range site.Languages }} {{- $lang := .Lang -}} {{- $isCurrent := eq $lang $currentLang -}} {{- $href := cond (eq $lang $defaultLang) "/" (printf "/%s/" $lang) -}} {{- with index (where $.Translations "Language.Lang" $lang) 0 -}} {{- $href = .RelPermalink -}} {{- end -}} {{- $aria := i18n (printf "lang_%s" (cond (eq $lang "en") "english" "danish")) -}} {{- if $isCurrent }} {{- partial "settings/flag.html" $lang -}} {{- else }} {{- partial "settings/flag.html" $lang -}} {{- end }} {{- end }}
